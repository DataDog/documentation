{{- $pattern := `^security_monitoring/default_rules/([a-zA-Z0-9]{3}-[a-zA-Z0-9]{3}-[a-zA-Z0-9]{3})$|^security/default_rules/([a-zA-Z0-9]{3}-[a-zA-Z0-9]{3}-[a-zA-Z0-9]{3})$` -}}
{{- range $p := .Site.Pages -}}
    {{- range .Aliases -}}
        {{- /* Normalize by replacing multi slashes with one */ -}}
        {{- $temp := (replaceRE `(?m)(\/\/+)` "/" .) -}}
        {{- $temp = strings.TrimPrefix "/" $temp -}}
        {{- $alias := strings.TrimSuffix "/" $temp -}}
        {{- if not (findRE $pattern $alias) -}}
            {{- $target := strings.TrimSuffix "/" $p.RelPermalink -}}
            {{- printf "%s/ %s\n" $alias $p.RelPermalink -}}
        {{- end -}}
    {{- end -}}
    {{- with $p.Params.external_redirect -}}
        {{- $redirect_link := . -}}
        {{- $temp := strings.TrimPrefix "/" $p.RelPermalink -}}
        {{- $target := strings.TrimSuffix "/" $temp -}}
        {{- if not (findRE $pattern $target) -}}
            {{- if not (strings.HasPrefix $redirect_link "https://") -}}
                {{- printf "%s/ %s\n" $target $redirect_link -}}
            {{- else -}}
                {{- printf "%s/ %s %s\n" $target $redirect_link "external" -}}
            {{- end -}}
        {{- end -}}
    {{- end -}}
{{- end -}}
{{- with .Site.Data.security_rules_global_aliases -}}
    {{- range . -}}
        {{- $alias := strings.TrimSuffix "/" (strings.TrimPrefix "/" .) -}}
        {{- printf "%s/ /security/default_rules/\n" $alias -}}
    {{- end -}}
{{- end -}}

{{- /* Removed rules have no Hugo page, so retain their URLs independently of page aliases. */ -}}
{{- $retiredRules := .Site.Data.security_rules_retired | default (dict) -}}
{{- range $site := .Sites -}}
    {{- $prefix := "" -}}
    {{- if ne $site.Language.Lang "en" -}}
        {{- $prefix = printf "%s/" $site.Language.Lang -}}
    {{- end -}}
    {{- $target := printf "/%ssecurity/default_rules/" $prefix -}}
    {{- range $id := $retiredRules.removed -}}
        {{- $aliases := index ($retiredRules.aliases | default (dict)) $id | default (slice) -}}
        {{- range $name := union (slice $id) $aliases -}}
            {{- range $path := slice "security/default_rules" "security_monitoring/default_rules" "security_platform/default_rules" "cloud_siem/default_rules" -}}
                {{- printf "%s%s/%s/ %s\n" $prefix $path $name $target -}}
                {{- printf "%s%s/%s.md %s\n" $prefix $path $name $target -}}
            {{- end -}}
        {{- end -}}
    {{- end -}}
{{- end -}}
